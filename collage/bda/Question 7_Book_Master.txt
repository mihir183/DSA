5. Use map reduce function to count total number of books published by each publisher.

var map = function() {
    emit(this.publisher, { count: 1 }); 
};

var reduce = function(key, values) 
{
    var total = 0; 
    for (var i = 0; i < values.length; i++) 
      {
        total += values[i].count;    
     }
    return { count: total }; 
}


db.Books.mapReduce(
    map,
    reduce,
    { out: "total_publishers" } // Output collection name
);
