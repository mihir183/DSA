11. Use map reduce function to count total quantity of each product.

var map = function()
 {
                var key = this.productID
            var value = {qty:this.saleQuantity}
            emit(key,value)
 }

 var reduce = function(key,value)
 {
    var reduce_val = { qty : 0 };
    for(var i=0;i<value.length;i++)
    {
            reduce_val.qty  =  reduce_val.qty + value[i].qty;
     }
     return reduce_val;
 }    

db.Sales.mapReduce(map,reduce,{out:"tot_qty"});

db.tot_qty.find().count();
